\framesubtitle{Summary}


\footnotesize
\begin{textblock*}{400pt}(30pt,70pt)
\begin{itemize}
\item Study investigates how tunneling ionization helps leverage 
multiphoton excitation to achieve inversion of populations in ionic gases
\item The target is ${\rm N}_2^+$ obtained from molecular ${\rm N}_2$ 
and ionized  with a strong field
\item The researchers showed that by exploiting technics as phase-locking
and stimulated emission, we can maximize ionic the lasing of the target
by one order of magnitude
\item This is another study deepening our understanding of strong field optics
\end{itemize}
\end{textblock*}

\framesubtitle{Theoretical insight}


\footnotesize
\begin{textblock*}{400pt}(30pt,70pt)
\begin{itemize}
\item From Lindbald theory, used to describe the dynamics of open systems, 
we write
\begin{equation}
\frac{d\rho}{dt} = -i\left[ H, \rho\right] 
 + \left[ \sum \gamma \left(A\rho A^\dagger 
 - \frac{1}{2}A^\dagger A\rho - \frac{1}{2}\rho A^{\dagger}A\right)\right],
\end{equation}
\noindent hastily reduced to, the following form
\begin{equation}
\frac{d\rho}{dt} = -i\left[ H, \rho\right] 
 + L(\rho)
\end{equation}
\item The Lindbald superopareator $L(\rho)$ meant to model the environmental 
conditions that make up the open quantum system
\item Here, it represents the population exchange from ionization and decay
not captured by the hermitian Hamiltonian, thus
\begin{equation}
\frac{d\rho^+}{dt} = -i\left[H(t),\rho^+ \right]
+ \left(\frac{d\rho^+_{n\nu}}{dt}\right)_{\rm ionize}
+ \left(\frac{d\rho^+_{n\nu,n^\prime\nu^\prime}}{dt}\right)_{\rm decay}
\end{equation}
\item Here we will be interested in $\Delta n = \rho^+_{B0,B0}-\rho^+_{X0,X0}$
\end{itemize}
\end{textblock*}
 
\end{frame}
