\frametitle{Femtosecond Reduction of Atomic Scattering Factors Triggered by
Intense X-Ray Pulse \small (Phys. Rev. Lett. \textbf{131}, 163201 (2023))}
\framesubtitle{Conclusion}


\small
\begin{textblock*}{400pt}(30pt,60pt)
\begin{itemize}
\footnotesize
\item XRD intensity of $\rm Si$ under irradiation from $11.5\,\rm keV$ 
XFEL pulse has been measured
\item Measurements reveal a diffraction intensity suppression 
for x-ray pulse intensity of $\sim 10^{19}\,\rm W/cm^2$ 
\item Data sugget that the diffraction intensity suppression is independent
from the scattering vector $\boldsymbol{Q}$
\item Thus, the cause has been determined to be in the change of individual
atom atomic scattering factors $f(\boldsymbol{Q})$ due to x-ray
electron excitation
\item The highly ionized state of the target when irradiated at that intensity
confirms the likeliness of the cause of the reduction to be 
the breakdown of the medium during the second half of the pulse

\end{itemize}
