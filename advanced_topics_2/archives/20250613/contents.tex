\item What is a Poincar\'e field?
\begin{itemize}
\footnotesize
\item[$\bullet$] Nonstandard term 
refering to fields that comply to the symmetries 
of the Poincar\'e group
\item[$\bullet$] The Poincar\'e group is the group of isometries of 
the Minkowski spacetime \\
(translations, rotations and boosts)
\end{itemize}
\item The notation ${\rm {\boldsymbol{A}}}_{jm}$, and subsequently $(j,m)$,
applied to the field thus gives some information about
\begin{itemize}
\footnotesize
\item[$\bullet$] the number of poles, specified by $j$
\item[$\bullet$] their spatial alignment in respect 
to the arbitrary $z$-axis $m$
%\begin{equation}
%\boldsymbol{A}(r,t) = \sum_{j,m}
%\left( a_{jm}^{(E)}\boldsymbol{A}_{jm}^{(E)}(\boldsymbol{r}) 
%+ a_{jm}^{(M)}\boldsymbol{A}_{jm}^{(M)}(\boldsymbol{r})\right)
%\end{equation}
\end{itemize}
\vspace{10pt}
%\item The authors use them because they allow to synchronously produce
%photoelectrons of both helicity
\item Spin-polarization refers to the bias 
in the orientation of spins in a beam of particle 
relative to a chosen direction
\begin{itemize}
\footnotesize
\item[$\bullet$] It's a statistical measure
\item[$\bullet$] not to be confused 
with the projection $m_s$ of the spin of a particle, which is intrinsic
\end{itemize}
\end{itemize}
\end{textblock*}

\framesubtitle{Atomic and field potentials}


\footnotesize
\begin{textblock*}{400pt}(30pt,70pt)
\begin{itemize}
\item The system is described by the Dirac equation
\begin{equation}
\Bigg( \gamma_\mu \left( \hat{p}^{\mu} + \left| e \right|A^{\mu}\right)
 - m_e\Bigg)\left|\Psi\right\rangle = 0 \label{eqn:dirac}
\end{equation}
where $|\Psi\rangle$ is the bispinor wavefunction, 
$\gamma^{\mu}$ are the Dirac matrices, and
$p^{\mu}$ and $A^{\mu}$ are the four-momentum and potential
\begin{align}
A^0(\boldsymbol{r}) & = V(r) = Z\alpha/\sqrt{|\boldsymbol{r}|^2+\delta r^2},
\hspace{20pt} \delta r \longrightarrow 0, \\
\boldsymbol{A}_{jm}(\boldsymbol{k},t) & = 
{\rm Re}\,A_0{\boldsymbol{\Phi}}_{jm}
\left( \theta_{\boldsymbol{k}},\varphi_{\boldsymbol{k}}\right)
\delta(|\boldsymbol{k}|-\omega)e^{i\omega t}, \\
{\rm where}\hspace{20pt} 
{\boldsymbol{\Phi}}_{jm}
\left( \theta_{\hat{\boldsymbol{k}}},\varphi_{\hat{\boldsymbol{k}}}\right) & =
\hat{\boldsymbol{k}}\times \boldsymbol{\nabla}_{\hat{\boldsymbol{k}}} Y_{jm}
\left( \theta_{\hat{\boldsymbol{k}}},\varphi_{\hat{\boldsymbol{k}}}\right), 
\hspace{40pt} \hat{\boldsymbol{k}} = {\boldsymbol{k}}/\omega
\end{align}
\vspace{20pt}
\begin{equation}
%j=\ell+m_s = 1/2 \hspace{40pt}\longrightarrow 
%\hspace{40pt}
j\geqslant 1, \hspace{30pt} |m| < j-1
\end{equation}
\end{itemize}
\end{textblock*}

\framesubtitle{Calculation of helicity}


\footnotesize
\begin{textblock*}{400pt}(30pt,70pt)
\begin{itemize}
\item The ion is initially in its ground state, written as $j=\ell + m_s=1/2$
\item The wavefunction $\left|\psi\right\rangle$ is 
obtained from (\ref{eqn:dirac})
using a finite-difference scheme
\item From the resulting wavefunction, we write
\begin{equation}
\left| \psi\right\rangle = 
\left|\psi_b\right\rangle + 
|\widetilde{\psi}\rangle
\end{equation}
\begin{itemize}
\footnotesize
\item[$\bullet$] Bound part $|\psi_b\rangle$ is filtered through the masking 
\begin{equation}
\mathcal{M}_\mu(\boldsymbol{r}) = 1-\exp\left(-|\boldsymbol{r}|^4/\mu^4\right)
\end{equation}
\item[$\bullet$] The resulting wavefunction is Fourier transformed
\item[$\bullet$] Low energy photoelectrons are then masked through
$\mathcal{M}_\mu(\boldsymbol{p})$
\end{itemize}
\item Helicity density is expressed as
\begin{equation}
\zeta(\boldsymbol{p}) = \langle \widetilde{\psi}(\boldsymbol{p}) |
\left( \boldsymbol{n}\cdot\frac{1}{2}\boldsymbol{\Sigma}\right)
|\widetilde{\psi}(\boldsymbol{p})\rangle, \hspace{30pt}
\boldsymbol{\Sigma}=\boldsymbol{\sigma}\mathds{1}, \hspace{20pt}
\boldsymbol{n} = \boldsymbol{p}/|\boldsymbol{p}|
\end{equation}
\end{itemize}
\end{textblock*}



\end{frame}
