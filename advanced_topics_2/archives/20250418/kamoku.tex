\begin{frame}
\frametitle{Multiphoton Ionization Reduction of Atoms in Two-Color 
Femtosecond Laser Fields \small (Phys. Rev. Lett. \textbf{130}, 113201 (2023))}
\framesubtitle{Theory}


\footnotesize
\begin{textblock*}{400pt}(30pt,70pt)
\begin{equation}
i\frac{\partial}{\partial t}\Psi(\boldsymbol{r},t) = 
\left[ -\frac{\boldsymbol{\nabla}^2}{2} + V(r) - i\boldsymbol{\rm A}(t)\cdot\boldsymbol{\nabla}\right]\Psi(\boldsymbol{r},t)
\end{equation}
\begin{itemize}
\item 2 different systems are used, so $V(r)$ is one of the following
\begin{align}
V(r) & = -1/r\hspace{40pt}&\ \mbox{for $\rm H$} \\
V(r) & = -(1+1.231e^{-0.662r}-1.325re^{-1.236r}-0.231e^{-0.48r})/r
\hspace{40pt}&\mbox{for $\rm He$}
\end{align}
\item The interacting field is given by
\begin{align}
A(t) = f(t)\left[ -\frac{E_{\rm SH}}{\omega_{\rm SH}}\sin(\omega_{\rm SH} t)
     -\frac{E_{\rm FW}}{\omega_{\rm FW}}\sin(\omega_{\rm FW} t+\phi) \right],
\hspace{30pt}f(t)=\sin^2(\pi t/T_p),
\hspace{30pt}T_p=20\frac{2\pi}{\omega_{\rm SH}}.
\end{align}
\item $\phi$ is the relative phase between the two fields of frequency 
$\omega_{\rm FW}$ and $\omega_{\rm SH}$ 
\end{itemize}
\end{textblock*}
 
\end{frame}


\begin{frame}
\frametitle{Multiphoton Ionization Reduction of Atoms in Two-Color 
Femtosecond Laser Fields \small (Phys. Rev. Lett. \textbf{130}, 113201 (2023))}
\framesubtitle{Numerical representation}


\footnotesize
\begin{textblock*}{400pt}(30pt,70pt)
\begin{itemize}
\item The wavefunction $\Psi(\boldsymbol{r},t)$ is expanded in a basis of 
B-splines functions, i.e
\begin{equation}
\Psi(\boldsymbol{r},t) 
  = \sum_{n\ell m}^{\infty} c_{n\ell m}(t) \Phi_{n\ell m}(\boldsymbol{r}),
  \hspace{20pt}
\Phi_{n\ell m}(\boldsymbol{r}) = 
 R_{n\ell}(r)Y_{\ell m}(\hat{\boldsymbol{r}}), \hspace{20pt}
 R_{n\ell}(r) = \sum_{i=1}^{N_r} d_{i}^{(n\ell)} B_i (r)
\end{equation}
\item The size of the box is $r_{\rm max}=1500\,\rm a.u.$, 
the time step is $\Delta t =0.008\,{\rm a.u.}$.
\item The total excitation probability is given by
\begin{equation}
P_{\rm excit} = \sum_{\substack{n\ell m \\ n>2}} 
\left|\left\langle \Phi_{n\ell m}^{\rm bound}(\boldsymbol{r}) | 
   \Psi(\boldsymbol{r},T_p) \right\rangle\right|^2.
\end{equation}

\item The total ionization probability is given by
\begin{equation}
P_{\rm ioniz} =  1 - \sum_{\substack{n\ell m}} 
\left|\left\langle \Phi_{n\ell m}^{\rm bound}(\boldsymbol{r}) | 
   \Psi(\boldsymbol{r},T_p) \right\rangle\right|^2.
\end{equation}

\item Convergence is achieved for a number of partial waves 
$\ell_{\rm max}=40$ and $N_r=1500$.
\end{itemize}
\end{textblock*}



\end{frame}
