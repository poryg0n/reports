\section*{About the nonlinear term}
The problem we are trying to solve is written as \cite{yangaliev2020}

\begin{equation}
	i\frac{\partial}{\partial t}\phi_\omega(x,t)
	   = \left( \hat{H}_a + \hat{V}_{\rm ext}(t)\right) 
	   \phi_\omega (x,t)
 	   + e^{i\omega t} {\hat{p}}\,
	   \psi(x,t), \label{eqn:main_1d}
\end{equation}


where

\begin{equation}
  \hat{H}_a = -\frac{1}{2}\frac{d^2}{dx^2} - \varkappa\delta(x), 
\hspace{20pt}  \hat{V}_{\rm ext} = F(t)x,
\end{equation}
\begin{equation}
  \phi_\omega (x,0) = \frac{-i\varkappa^{3/2}}{\omega}
{\rm sgn}(x)\left(e^{-\varkappa|x|} - e^{-\varkappa(\omega)|x|}\right), 
\hspace{20pt}  \psi(x,0) = \varkappa^{1/2}e^{-\varkappa|x|}, 
\label{eqn:bound_state}
\end{equation}

with

\begin{equation}
  \hat{p} = -id/dx,\hspace{10pt} \mbox{ and } 
  \hspace{10pt} \varkappa(\omega) = \sqrt{\varkappa^2 + 2\omega}.
\end{equation}

\hspace{10pt}

A RK4 adapted to our problem has been written and must be tested as 
rigourously as possible.


Let us first consider the simpler following problem

\begin{equation}
	i\frac{\partial}{\partial t}\psi(x,t)
	   = \left( \hat{H}_a + \hat{V}_{\rm ext}(t)\right) 
	   \psi(x,t),   \hspace{20pt}\psi(x,0)=\psi_0(x),  \label{eqn:step_1d}
\end{equation}
\begin{equation}
  \hat{H}_a = -\frac{1}{2}\frac{d^2}{dx^2} - \varkappa\delta(x), \hspace{20pt}
  \hat{V}_{\rm ext} = F(t)x,
\end{equation}
\begin{equation}
  E_0 = -\varkappa^2/2,\hspace{20pt}
  \psi_{0}(x) = \varkappa^{1/2}e^{-\varkappa|x|}, \hspace{20pt}
  \varkappa = 1       \label{eqn:bound_state_s}
\end{equation}
\begin{equation}
  E_k = k^2/2,\hspace{20pt}
  \psi_k^{(\pm)}(x) = e^{ikx} + R(\pm|k|)e^{\pm i|kx|}, \hspace{20pt}
   R(k) = \frac{i\varkappa}{k-i \varkappa}. \label{eqn:inout_states}
\end{equation}
\vspace{10pt}
The parameters of the laser we will be using 
are given by the relations
\begin{equation}
  F(t) = F_0 e^{-(2t^\prime/\tau)^2}\cos(\omega_0 t^\prime),
\hspace{20pt}  0 \leqslant t \leqslant T,
\hspace{20pt}  t^{\prime} = t - T/2,
\end{equation}
\begin{equation}
  \tau  = 2\pi n_{oc}/\omega_0,  \hspace{20pt}
  n_{oc} = 5. \hspace{20pt}
\end{equation}

Ultimately, Eq. (\ref{eqn:step_1d}) is of the form 

\begin{equation}
 \dot{y} = -i A_{n\times n}\,y,
\end{equation}

where the dot denotes differentiation with respect to time and $A$
is a square, real and hermitician matrix. Of course, given the problem, the 
solutions $y$ are nonetheless complex.

\section{Detailed formulation of the propagator}

Following Kidd \textit{et al.} \cite{kidd2017}, we want to write 
the problem (\ref{eqn:main_1d}) as 

\begin{align}
\frac{dy}{dt} = f(y,t), \hspace{20pt} y(t=0) = y_0, \\
f(y,t) = \mathcal{L}y + \mathcal{N}(y,t),
\end{align}

\noindent where $\mathcal{L}$ is the linear part. 
$\mathcal{N}(y,t)$ is a priori a nonlinear term, 
but we will use it as a nonhomogeneous term in the following. \\

Let us consider the Schr\"odinger equation
\begin{align}
i\frac{\partial}{\partial t} \Psi(x,t) & = H(x,t) \Psi(x,t), \\
           & = \mathcal{L}(x,t)\Psi(x,t) + \mathcal{N}(\Psi,t),
\end{align}

\noindent with $\mathcal{L} = H_0 + V_{\rm ext}$, 
where $H_0$ is the unperturbed Hamiltonian and $V_{\rm ext}$,
is the interaction with the laser field. 
In the following the inhomogeneous term 
will be considered to assume the following form

\begin{equation}
\mathcal{N}(\Psi,t) = \mathcal{N}\Psi(x,t).
\end{equation}

Using the integrating factor method, the solution will be probed
under the form

\begin{equation}
 \Phi(x,t) = e^{i\mathcal{L}t}\Psi(x,t). \label{eqn:if}
\end{equation}

Using this in the Eq. (\ref{eqn:step_1d}), we get
\begin{align}
  i\frac{\partial}{\partial t}\left( e^{-i\mathcal{L}t}\Phi(x,t) \right)
         & = i\left( -i\mathcal{L}e^{-i\mathcal{L}t}\Phi(x,t)
          + e^{-i\mathcal{L}t}
          \frac{\partial}{\partial t}\Phi(x,t)  \right) \\
         & =  \mathcal{L}e^{-i\mathcal{L}t}\Phi(x,t)
          + ie^{-i\mathcal{L}t}
          \frac{\partial}{\partial t}\Phi(x,t)
\end{align}

Equating with the RHS, we then get

\begin{align}
     \frac{\partial}{\partial t}\Phi(x,t)
         & = - ie^{-i\mathcal{L}t}
          \mathcal{N}\left( e^{-i\mathcal{L}t} \Phi, t\right).
\end{align}

Adopting the notation $f(x,t_n) = f_n$, and integrating between $t_n$
and $t_{n+1}$, we get 

\begin{align}
     \Phi_{n+1} & = \Phi_{n} - i\Delta t\,e^{-i\mathcal{L}t_n}
          \mathcal{N}\left( e^{-i\mathcal{L}t_n} \Phi_n, t_n\right).
\end{align}

From Eq. (\ref{eqn:if}), and setting $\Delta t = {t_{n+1}-t_n}$, 
we then get

\begin{align}
     \Psi_{n+1} & = e^{-i\mathcal{L}\Delta t}\Psi_{n} 
          - i\Delta t\,e^{-i\mathcal{L}\Delta t} 
             \mathcal{N}\left(\Psi_n, t_n\right).
  \label{eqn:main_problem}
\end{align}

This last equation is the formulation of our main problem.
When the inhomogeneous term is set to the null matrix, 
Eq. (\ref{eqn:main_problem}) reduces to the split-operator method. 
When the linear part is considered null, which amounts to setting 
$H = \mathcal{N}$, we have a full Runge-Kutta scheme.
For the sake of evalauting the precision of our RK scheme, 
this last situation is the exact one we want to be in.


In the following, we set the initial propagation time $t_i = 0 $ and
the final propagation time $t_f = 2\pi$. The time step is then defined 
as $\Delta t = \left(t_f - t_i\right)/N$, where $N$ is the number of  
propagation steps. For the sake of assessing the accuracy of our scheme, 
we consider the solutions obtained from the split-operator method 
as the exact solution.
We define the relative error between the exact solution 
and the numerical solution (obtained from our Runge-Kutta scheme) as 
\begin{equation}
\sigma_r(j) = \left|\frac{y(j) - y_{\rm ex}(j)}{y_{\rm ex}(j)}\right|
\label{eqn:rel_err}
\end{equation}

\section{Test 2}

We now want to test our scheme by setting 
\begin{equation}
\mathcal{L} = 0, \hspace{20pt}
\mathcal{N} = \left(
\begin{matrix}
 1 & 2 \\
 2 & 1 
\end{matrix}
\right).
\end{equation}

The problem is therefore the following ODE system
\begin{equation}
  \Bigg\lbrace
\begin{matrix}
 y_1^{\prime} = -i y_1 -2i y_2  \\
 y_2^{\prime} = -i 2y_1 - i y_2 
\end{matrix}, \hspace{20pt}\mbox{ with}\hspace{20pt} y_{i}^0 = y_i(t=0) = i.
\label{eqn:test2}
\end{equation}

We can see that the eigenvalues of the matrix are $-1$ and
$3$, which makes the eigenvalues of this problem 
$r_1 = i$ and $r_2 = -3i$.
The solutions can be found by hand. They read
\begin{align}
 y_1 = -\frac{1}{2}e^{it} + \frac{3}{2}e^{-3it}\hspace{20pt}\mbox{and} 
 \hspace{20pt} 
 y_2 = \frac{1}{2}e^{it} + \frac{3}{2}e^{-3it} \label{eqn:test2}
\end{align}

We solve this problem by twice, using both the split-operator method and
Runge-Kutta and compare the results by the mean of (\ref{eqn:rel_err}).
The results are presented in Tab. (\ref{tab:test2}) and Fig. (\ref{fig:test2}).

\section{Test 3}

Let us now consider the case
\begin{equation}
\mathcal{L} = 0, \hspace{20pt}
\mathcal{N} = \left(
\begin{matrix}
 1   & 1 & 1 \\
 1   & 2 & 2 \\
 1   & 2 & 3 
\end{matrix}
\right).
\end{equation}

represented by following ODE system
\begin{equation}
  \Bigg\lbrace
\begin{matrix}
 y_1^{\prime} = -i y_1 -i y_2 - iy_3  \\
 y_2^{\prime} = -i y_1 -2i y_2 - 2iy_3  \\
 y_3^{\prime} = -i y_1 -2i y_2 - 3iy_3  
\end{matrix}, \hspace{20pt}\mbox{ with}\hspace{20pt} y_{i}^0 = y_i(t=0) = i.
\label{eqn:test3}
\end{equation}

This problem is numerically solved using Lapack. The eigenvalues are given by
$r_1 = -0.307978528369904i$, $r_2 = -0.643104132107790i$ and 
$r_3 = -5.04891733952230i$.
